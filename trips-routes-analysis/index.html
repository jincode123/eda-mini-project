This section examines trip and route characteristics to determine whether certain trips or routes are associated with higher incident risk. Factors such as trip length, frequency, and route-level patterns are explored.

The analysis focuses on identifying variations in incident rates across routes and trips to better understand how risk is distributed.

trips_df['dispatch_date']=pd.to_datetime(trips_df['dispatch_date'])
trips_df['year']=pd.to_datetime(trips_df['dispatch_date']).dt.year
trips_df['month_year']=pd.to_datetime(trips_df['dispatch_date']).dt.to_period('M')
yearly_trips = (
    trips_df.groupby('year')['trip_id']
    .count()
    .reset_index()
)

# 2. Create the bar chart
plt.figure(figsize=(5.5, 3))

# Use hue='year' and legend=False to avoid the FutureWarning
ax = sns.barplot(
    data=yearly_trips, 
    x='year', 
    y='trip_id', 
    hue='year', 
    palette='viridis', 
    legend=False
)

# 3. THE TRICK: Zoom in on the Y-axis to make differences obvious
# We set the bottom of the axis just below your lowest trip count
min_trips = yearly_trips['trip_id'].min()
max_trips = yearly_trips['trip_id'].max()
plt.ylim(min_trips * 0.95, max_trips * 1.05) 

# 4. Add data labels on top of the bars
for p in ax.patches:
    ax.annotate(format(p.get_height(), ',.0f'), 
                (p.get_x() + p.get_width() / 2., p.get_height()), 
                ha = 'center', va = 'center', 
                xytext = (0, 9), 
                textcoords = 'offset points')

plt.title('Yearly Trips Trend')
plt.xlabel('Year')
plt.ylabel('Total Trips')
plt.grid(axis='y', linestyle='--', alpha=0.7)
plt.show()
